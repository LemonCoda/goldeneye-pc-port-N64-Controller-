cmake_minimum_required(VERSION 3.16)
project(nso_n64_tests C)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake")
find_package(SDL2 REQUIRED)
add_executable(nso_n64_tests test_nso_n64.c ../../port/src/nson64.c)
target_include_directories(nso_n64_tests PRIVATE ../../port/include ${SDL2_INCLUDE_DIR})
target_link_libraries(nso_n64_tests PRIVATE ${SDL2_LIBRARY})
set_target_properties(nso_n64_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(nso_n64_tests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME nso_n64_mapping COMMAND nso_n64_tests)
